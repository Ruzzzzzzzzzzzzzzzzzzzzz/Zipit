Client setup

        
Enterprise license client configuration. For multi-software authorization use the authGate section; the server URL is filled in automatically.

        

          

            

              
license:
  enabled: true
  file: "license.yml"
  publicKey: "public.pem"
  server: "{{PUBLIC_URL}}"
  require: true
  features:
    - audit

            

            

          

          

            

              
authGate:
  enabled: true
  server: "{{PUBLIC_URL}}"
  software: "my-app"
  title: "My App"
  register: true
  cacheFile: ""
  timeoutMs: 5000
  entryPoints: []

            

            

          

        

        

          
--auth-gate --auth-server {{PUBLIC_URL}} --auth-software my-app --auth-title "My App"

        

        
Environment ZIPIT_LICENSE_SERVER={{PUBLIC_URL}}. Clients verify the RSA signature locally and only query this server for revocation, online validation and multi-software authorization.
